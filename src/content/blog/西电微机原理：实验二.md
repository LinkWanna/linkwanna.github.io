## 前言

这篇是西电计科专业微机原理随课实验的第四篇文章，到这里，还是更加复杂的汇编实验，我们需要实现**从终端输入任意位数的数字，然后再将其转换为二进制的形式输出到终端上，若用户输入为 `q` 或 `Q` 则退出程序**。前面的实验一已经把基础设施全部搭建完成了，这一次我们会直接使用之前的代码框架。


## 代码复用

我把需要复用的代码分别拆成了 `common.asm` 和 `debug.asm` 两个文件，前者是对 Dos 系统的调用的封装，后者是一些调试用的函数。我们只需要在新的实验文件中引用这两个文件即可。

### common.asm - 常用系统调用封装

```asm
; common.asm - 常用系统调用封装
; 使用 cdecl 调用约定进行传参，即传参规则如下：
; 1. 参数从右到左压入栈，调用者负责清理栈
; 2. 返回值通过 AX 寄存器返回

; 为了简化调用，需要保证所有的调用都是以字为单位
; Byte: 1 字节          u8
; Word: 2 字节          u16
; Double Word: 4 字节   u32

doscall macro dosno, funcno
    mov     ah, dosno
    mov     al, funcno
    int     21h
endm

; 获取一个字符到 AL
; void exit(u8 code);
exit proc
    push    bp
    mov     bp, sp
    mov     al, [bp + 4]       ; 获取退出代码
    
    mov ah, 4Ch
    int     21h

exit endp

; 获取一个字符到 AL
; u8 getc();
getc proc
    push    bp
    mov     bp, sp
    sub     sp, 1           ; 给 AH 留 1 字节
    mov     [bp - 2], ah    ; 保存 AH 

    doscall 01h, 00h

    ; 恢复栈空间
    mov     ah, [bp - 2]
    mov     sp, bp
    pop     bp
    ret
getc endp

; 输出一个字符
; void putc(u8 ch);
putc proc
    push    bp
    mov     bp, sp
    push    ax
    push    dx

    ; 从栈中获取用户输入的字符
    ; 2 个字节的返回地址 + 2 个字节的参数
    mov     dl, [bp + 4]
    doscall 02h, 00h

    pop     dx
    pop     ax
    pop     bp
    ret
putc endp

; 输出以 '$' 结尾的字符串，DS:DX 指向字符串
; void puts(u8* str);
puts proc
    push    bp
    mov     bp, sp
    push    ax
    push    dx
    
    mov     dx, [bp + 4]   ; 获取字符串地址
    doscall 09h, 00h

    pop dx
    pop ax
    pop bp
    ret
puts endp

; 获取用户输入的字符串，DS:DX 指向输入缓冲区
; void gets(u8* buf);
gets proc
    push    bp
    mov     bp, sp
    push    ax

    mov     dx, [bp + 4]   ; 获取输入缓冲区地址
    doscall 0Ah, 00h

    pop ax
    pop bp
    ret
gets endp

; 仅 8086 以上支持
; 函数调用宏，模拟 cdecl 调用约定
; INVOKE_C function, <arg[n], arg<n-1>, ..., arg2, arg1>
INVOKE_C macro function:req, args:vararg
    local argc, arg
    argc = 0

    ; 先数参数个数，便于后面清栈
    for arg, <args>
        argc = argc + 1
    endm

    ; 从左到右压参
    for arg, <args>
        push arg
    endm

    call function

    ; 调用者清栈（cdecl 风格）
    if argc gt 0
        add  sp, argc*2           ; 16 位下每参数 2 字节
    endif
endm
```

### debug.asm - 调试用函数

```asm
; debug.asm - 调试用函数

; 打印 0-9 A-F 的数字
; void putnum(const u8 num);
putnum proc
    push    bp
    mov     bp, sp
    push    ax

    mov     al, [bp + 4]        ; 获取参数 num
    cmp     al, 9
    jbe     @@print_num

    ; 16 进制 A-F
@@print_alpha:
    add     al, 'A' - 10
    jmp     @@exit

    ; 10 进制 0-9
@@print_num:
    add     al, '0'
    jmp     @@exit

@@exit:
    INVOKE_C putc, <ax>         ; 输出字符

    pop     ax
    pop     bp
    ret
putnum endp

; 需要输出的 ascii 码
; assert(radix > 1)
; void print(u16 ascii, u8 radix);
print proc
    push    bp
    mov     bp, sp
    push    ax
    push    bx
    push    dx
    push    si
    sub     sp, 8

    ; char data[16]             ; 保存余数的数组
    mov     si, 0               ; 数组索引

    ; dx:ax = ascii. bx = radix
    xor     dx, dx              ; 清除 dx，准备除法
    mov     ax, [bp + 4]        ; 设置被除数
    mov     bl, [bp + 6]        ; 设置除数
    xor     bh, bh              ; 清除高位以防干扰


    ; 循环除以 <radix>，直到商为 0
@@print_while_1:
    div     bx                  ; 除以 <radix>，商在 AX，余数在 DX

    ; 余数保存到局部变量
    mov     [bp + si - 8 - 8], dl  ; data[si] = dl
    inc     si                  ; 索引加 1
    xor     dx, dx              ; 清除 dx

    cmp     ax, 0
    jne     @@print_while_1

    ; 逆序输出保存的余数
@@print_while_2:
    dec     si
    mov     dl, [bp + si - 8 - 8]     ; 取出 dl = data[si]
    INVOKE_C putnum, <dx>
    cmp     si, 0
    jne     @@print_while_2

    ; 恢复栈
    add     sp, 8
    pop     si
    pop     dx
    pop     bx
    pop     ax
    mov     sp, bp
    pop     bp
    ret
print endp
```

## 5 位数字输入的实现

任意位数的数字输入比较复杂，我们先实现一个 5 位数字输入的版本，话说，为什么偏偏是 5 位呢？因为 5 位十进制数字的最大值是 99999，转换成二进制是 16 位以内的数（11000011010111111），如果是 6 位，如果它大于 65535，那就超出 16 位的范围了。
通过这样的简化，我们就可以直接用一个 16 位的变量来存储输入的数字了。

我们现在来整理一下思路，用户会输入一个字符串，我们需要把这个字符串转换成数字，然后再把数字转换成二进制字符串输出。简单整理一下：
1. 从终端获取用户输入的字符串
2. 检查用户是否输入了 `q` 或 `Q`，如果是则退出程序
3. 把数字字符串转换成一个数字数组
4. 根据不同位数上的权值把数字数组转换成一个 16 位的数字
5. 输出二进制字符串到终端

获取用户输入可以直接使用 `common.asm` 里面的 `gets` 函数，检查用户的输入可以从《实验一》中找到相关代码。输出二进制字符串也可以使用 `debug.asm` 里面的 `print` 函数，我们只需要把 `radix` 参数设置为 2 即可。最后留给我们的工作就是把数字字符串转换成数字数组，然后再把数字数组转换成一个 16 位的数字了。

不过，在实现之前，我们先来实现两个辅助函数：`char2num` 和 `multen`。
1. `u8 char2num(const u8 ch)`：把一个字符 `ch` 转换成对应的数字
2. `u16 multen(const u8 num, const u8 times)`：计算 `num * (10 ^ times)` 的值

### 字符转换成数字(char2num)

这个函数只需要把字符的 ASCII 码减去字符 `0` 的 ASCII 码即可得到对应的数字。不过除此之外，我们还需要验证用户输入的字符是否在 `0` 到 `9` 的范围内，如果不在这个范围内，则提示用户输入了无效字符并退出程序。

```asm
; 将字符数字转换为数值
; u8 char2num(const u8 ch);
char2num proc
    push    bp
    mov     bp, sp

    mov     al, [bp + 4]        ; 获取字符参数
    ; 验证字符串是否在 0~9 范围内
    cmp     al, '0'
    jl      @@invalid_char
    cmp     al, '9'
    jg      @@invalid_char

    sub     al, '0'             ; 转换为数字
    mov     ah, 0                ; 清除高位
    jmp     @@char2num_exit

    ; 无效字符处理
@@invalid_char:
    INVOKE_C puts, <offset invid>
    INVOKE_C puts, <offset crlf>
    INVOKE_C exit, 1
    
@@char2num_exit:
    pop     bp
    ret
char2num endp

; 测试代码
mov    al, '1'
INVOKE_C char2num, <ax>
INVOKE_C print, 10, <ax>

mov    al, '5'
INVOKE_C char2num, <ax>
INVOKE_C print, 10, <ax>

mov    al, '9'
INVOKE_C char2num, <ax>
INVOKE_C print, 10, <ax>

; 无效字符测试
mov    al, 'A'
INVOKE_C char2num, <ax>
INVOKE_C print, 10, <ax>
```

### 乘以 10 的幂(multen)

这个函数需要计算 `num * (10 ^ times)` 的值，我们可以通过循环来实现这个功能。

```asm
; 将 num 乘以 10 的 times 次方，并返回结果
; u16 multen(const u8 num, const u8 times);
multen proc
    push    bp
    mov     bp, sp
    push    bx
    push    cx
    push    dx  ; 保存 dx，因为 mul 指令会用到 dx

    mov     al, [bp + 4]        ; 获取 num
    mov     cx, [bp + 6]        ; 获取 times

    mov     ah, 0                ; 清除高位
    mov     bx, 10               ; 乘数 10

    cmp     cx, 0
    je      @@multen_end
@@multen_loop:
    mul     bx                   ; 结果乘以 10
    loop    @@multen_loop

@@multen_end:
    pop     dx
    pop     cx
    pop     bx
    pop     bp
    ret
multen endp

; 测试代码
mov     ax, 2
mov     bx, 2
INVOKE_C multen, <bx>, <ax>
INVOKE_C print, 10, <ax>      ; 输出 200

mov     ax, 2
mov     bx, 4
INVOKE_C multen, <bx>, <ax>
INVOKE_C print, 10, <ax>      ; 输出 20000

mov     ax, 2
mov     bx, 0
INVOKE_C multen, <bx>, <ax>
INVOKE_C print, 10, <ax>      ; 输出 2
```

### 完整实现

```asm
space   db      ' ', '$'
prompt  db      'Input a number: $'
invid   db      'Invalid input!$'

num_input  db 20          ; byte 0: 最大输入长度
           db ?           ; byte 1: 实际输入长度
           db 20 dup(0)   ; byte 3: 输入缓冲区

.code
include common.asm
include debug.asm

; 将 num 乘以 10 的 times 次方，并返回结果

    push    bx
    push    cx
    push    dx

    mov     al, [bp + 4]        ; 获取 num
    mov     cx, [bp + 6]        ; 获取 times

    mov     ah, 0                ; 清除高位
    mov     bx, 10               ; 乘数 10

    cmp     cx, 0
    je      @@multen_end
@@multen_loop:
    mul     bx                   ; 结果乘以 10
    loop    @@multen_loop

@@multen_end:
    pop     dx
    pop     cx
    pop     bx
    pop     bp
    ret
multen endp

; 将字符数字转换为数值
; u8 char2num(const u8 ch);
char2num proc
    push    bp
    mov     bp, sp

    mov     al, [bp + 4]        ; 获取字符参数
    ; 验证字符串是否在 0~9 范围内
    cmp     al, '0'
    jl      @@invalid_char
    cmp     al, '9'
    jg      @@invalid_char

    sub     al, '0'             ; 转换为数字
    mov     ah, 0                ; 清除高位
    jmp     @@char2num_exit

    ; 无效字符处理
@@invalid_char:
    INVOKE_C puts, <offset invid>
    INVOKE_C puts, <offset crlf>
    INVOKE_C exit, 1
    
@@char2num_exit:
    pop     bp
    ret
char2num endp

main proc
    ; 初始化数据段和栈指针
    mov     ax, @data
    mov     ds, ax

@@main_while_1:
    ; 1. 获取用户输入的数字
    INVOKE_C puts, <offset prompt>
    INVOKE_C gets, <offset num_input>
    INVOKE_C puts, <offset crlf>
    mov     si, offset num_input

    ; 2. 首先判断输入是否为 `q` 或 `Q`, 以退出程序
    mov     al, [si + 2]   ; 获取第一个字符
    cmp     al, 'q'
    je      @@end
    cmp     al, 'Q'
    je      @@end

    ; 3. 获取实际输入长度，并判断是否为空
    mov     cx, 0           ; 清除 CX
    mov     cl, [si + 1]    ; CX = 实际输入长度
    cmp     cx, 0           ; 用户输入空字符串，则跳过计算
    je      @@main_while_2_end

    ; 4. 根据位数进行加和，得到一个 16 位的数字
    mov     dx, 0           ; 清除结果寄存器 DX
    add     si, 2           ; 指向实际输入的第一个字符
@@main_while_2:
    ; 4.1 将输入的字符串转换为数字
    mov     ah, 0
    mov     al, [si]       ; 获取当前字符
    inc     si             ; 位数索引加 1
    INVOKE_C char2num, <ax>

    ; 4.2 计算该数字对应的权值，并加到结果中
    mov     bx, cx          ; 保存当前位数
    dec     bx              ; 位数 - 1
    INVOKE_C multen, <bx>, <ax>

    ; 4.3 将结果累加到 DX 中
    add     dx, ax
    loop    @@main_while_2

@@main_while_2_end:
    ; 5. 输出结果
    INVOKE_C print, 2, <dx>
    INVOKE_C puts, <offset crlf>
    jmp    @@main_while_1


@@end: 
    INVOKE_C puts, <offset bye>
    ; 带返回码 0 结束
    mov     ah, 4Ch         
    xor     al, al
    int     21h

main endp
end main
```

## 任意位数字输入的实现（进阶）

在上面的实现中，我们只能处理 5 位以内的数字输入，如果用户输入的数字超过了 5 位，那么就会出现溢出的情况。虽然不会让程序 crash，但输出的结果肯定是错误的。
为了支持任意位数的数字输入，我们需要使用一个字节数组来存储输入的数字，然后通过不断地除以 2 来实现十进制转二进制的功能。本质上就是模拟大数运算。

下面的一个 Python 的实现示例，展示了如何将任意长度的十进制数字字符串转换为二进制字符串：
```python
# 字符串模拟大数运算
# 将任意长度的 10 进制数转换为 2 进制并进行打印
dec_num = "1234"

def dec_to_bin(dec_str):
    if dec_str == "0":
        return "0"
    
    bin_str = ""
    dec_list = [int(d) for d in dec_str]

    # 不断除以 2，记录余数
    while dec_list:
        remainder = 0
        new_dec_list = []
        # 模拟除法过程
        for digit in dec_list:
            # 计算当前位与余数结合后的新值
            current = remainder * 10 + digit
            new_digit = current // 2
            remainder = current % 2
            # 只添加非零位，避免前导零
            if new_dec_list or new_digit != 0:
                new_dec_list.append(new_digit)
        # 记录余数
        bin_str = str(remainder) + bin_str
        dec_list = new_dec_list

    return bin_str

binary_num = dec_to_bin(dec_num)
print(f"Decimal: {dec_num}")
print(f"Binary: {binary_num}")
```

### 汇编实现

细节比较多，我们直接上代码，下面的代码需要注意几点：
1. 栈的使用：为了避免栈溢出，我们需要在栈上分配足够的空间来存储中间结果。
2. 大数除法的实现：我们需要实现一个函数 `div2_byte_array` 来模拟数字除以 2 的过程，并返回余数。

```asm
.286
.model small
.stack 200h

.data
crlf    db      13, 10, '$'
space   db      ' ', '$'
prompt  db      'Input a number: $'
invid   db      'Invalid input!$'
bye     db      'GoodBye!' ,'$'

num_input  db 64          ; byte 0: 最大输入长度
           db ?           ; byte 1: 实际输入长度
           db 64 dup(0)   ; byte 3: 输入缓冲区

.code
include common.asm
include debug.asm

; 将字符数字转换为数值
; u8 char2num(const u8 ch);
char2num proc
    push    bp
    mov     bp, sp

    mov     al, [bp + 4]        ; 获取字符参数
    cmp     al, '0'
    jl      @@invalid_char
    cmp     al, '9'
    jg      @@invalid_char

    sub     al, '0'             ; 转换为数字
    mov     ah, 0                ; 清除高位
    jmp     @@char2num_exit

@@invalid_char:
    INVOKE_C puts, <offset invid>
    INVOKE_C puts, <offset crlf>
    INVOKE_C exit, 1
    
@@char2num_exit:
    pop     bp
    ret
char2num endp

; 大整数除以 2
; u16 div2_byte_array(u8* array, u16 len);
; array: 指向字节数组的指针
; len: 数组长度（字节数）
; return: al = 余数, ax = len 是否减一
; 数组按大端存储，即高位在前，低位在后
div2_byte_array proc
    push    bp
    mov     bp, sp
    push    bx
    push    cx
    push    dx
    push    si
    push    di
    sub     sp, 1

    ; is_leading_zero = false
    mov     byte ptr [bp - 10 - 1], 0

    ; 从栈上获取参数
    mov     si, [bp + 4]        ; 获取 array 指针
    mov     cx, [bp + 6]        ; 获取 len
    mov     di, si              ; DI 用于遍历数组
    mov     dx, 0               ; 进位标志，初始为 0
    mov     ax, 0               ; 清除 AX

@@div2_loop:
    ; 处理进位
    mov bl, 10  ; 乘数
    mov al, dl  ; 进位放到 AL 的高位
    mul bl      ; AX = DL * 10, 可以确保 AH 始终为 0
    
    ; 把当前字节加上进位后的值放到 AL
    ; 并处理进行除法
    add     al, [si]
    inc     si                  ; 移动到下一个字节

    mov     bl, 2               ; 除数
    div     bl                  ; AL = AL / 2，AH = AL % 2
    mov     dl, ah              ; 保存余数到 dl，作为下一个字节的进位
    
    ; 避免前导 0 出现
    ; 若 di 指向第一个字节，则需要检查前导 0
    cmp     di, [bp + 4]
    jne     @@save_result

@@check_leading_zero:
    cmp     al, 0
    je     @@skip_leading_zero

@@save_result:
    mov     [di], al            ; 保存结果回数组
    inc     di 
    jmp     @@done_check_leading_zero

@@skip_leading_zero:
    mov     byte ptr [bp - 10 - 1], 1   ; 已跳过前导 0

@@done_check_leading_zero:
    ; 不处理前导零,就两行代码
    ; mov     [di], al            ; 保存结果回数组
    ; inc     di                  ; 移动到下一个字节
    loop    @@div2_loop

    ; 设置返回值
    mov     al, dl              ; 余数放到 AL
    mov     ah, [bp - 10 - 1]        ; 高位标志放到 AH

    add     sp, 1
    pop     di
    pop     si
    pop     dx
    pop     cx
    pop     bx
    mov     sp, bp
    pop     bp
    ret
div2_byte_array endp


; 十进制转二进制,并打印
; void dec2bin(u8* dec, u8* bin, u16 len);
dec2bin proc
    push    bp
    mov     bp, sp
    push    ax
    push    cx
    push    dx
    push    si
    push    di

    sub     sp, 256

    ; 从栈上获取参数
    mov     si, [bp + 4]        ; 获取 dec 指针
    mov     cx, [bp + 6]        ; 获取 len
    mov     di, 0               ; 获取 bin 指针


    ; 清除 DX
    xor     dx, dx

@@dec2bin_loop:
    ; 1. 将当前十进制数不断除以 2，直到结果为 0
    INVOKE_C div2_byte_array, <cx>, <si>
    
    ; 2. 获取余数，存入二进制数组
    mov     [bp + di - 10 - 256], al    ; 存储余数
    inc     di          ; 移动到下一个存储位置
    
    ; 3. 检查是否需要减少 len
    cmp     ah, 1
    jne     @@no_len_dec
    dec     cx

@@no_len_dec:
    ; 退出条件: len == 0
    cmp     cx, 0
    jne      @@dec2bin_loop

    ; 4. 逆序打印二进制数组
@@print_loop:
    dec     di
    mov     al, [bp + di - 10 - 256]
    INVOKE_C putnum, <ax>
    cmp     di, 0
    jne     @@print_loop

    pop     di
    pop     si
    pop     dx
    pop     cx
    pop     ax
    mov     sp, bp
    pop     bp
    ret
dec2bin endp

main proc
    ; 初始化数据段和栈指针
    mov     ax, @data
    mov     ds, ax

@@main_while_1:
    ; 1. 获取用户输入的数字
    INVOKE_C puts, <offset prompt>
    INVOKE_C gets, <offset num_input>
    INVOKE_C puts, <offset crlf>

    mov     si, offset num_input
    ; 2. 首先判断输入是否为 `q` 或 `Q`, 以退出程序
    mov     al, [si + 2]   ; 获取第一个字符
    cmp     al, 'q'
    je      @@end
    cmp     al, 'Q'
    je      @@end

    ; 3. 获取输入的数字字符串
    mov     bx, offset num_input

    ; 获取实际输入长度
    xor     cx, cx
    mov     cl, [bx + 1]        ; CX = 实际输入长度
    xor     si, si              ; SI 指向第一个字符
    
    cmp     cx, 0
    je      @@main_while_2_end
@@main_while_2:
    ; 4. 将输入的字符串转换为数字
    mov     ah, 0
    mov     al, [bx + si + 2]    ; 获取当前字符
    INVOKE_C char2num, <ax>

    ; 5. 将数字存储回去
    mov     [bx + si + 2], al
    inc     si          ; 位数索引加 1
    
    loop    @@main_while_2
@@main_while_2_end:
    ; 6. 调用 dec2bin 转换并打印
    xor    cx, cx
    mov     cl, [bx + 1]        ; CX = 实际输入长度
    INVOKE_C dec2bin, <cx>, <offset num_input + 2>
    INVOKE_C puts, <offset crlf>

    jmp    @@main_while_1
@@end: 
    INVOKE_C puts, <offset bye>
    ; 带返回码 0 结束
    mov     ah, 4Ch         
    xor     al, al
    int     21h
main endp
end main
```
